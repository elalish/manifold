109inline double sind(double x) {
110 if (!la::isfinite(x)) return NAN;
111 if (x < 0.0) return -sind(-x);
112#if defined(_WIN32) && defined(_M_ARM64)
113 x = std::fmod(x, 360.0);
114 const int quo = static_cast<int>(std::floor(x / 90.0 + 0.5));
115 x -= quo * 90.0;
116#else
117 int quo;
118 x = std::remquo(std::fabs(x), 90.0, &quo);
119#endif
120 const double xr = radians(x);
121 switch (quo % 4) {
122 case 0:
123 return math::sin(xr);
124 case 1:
125 return math::cos(xr);
126 case 2:
127 return -math::sin(xr);
128 case 3:
129 return -math::cos(xr);
130 }
131 return 0.0;
132}

304struct Box {

306 vec3 max = vec3(-std::numeric_limits<double>::infinity());

311 constexpr Box() {}

316 constexpr Box(const vec3 p1, const vec3 p2) {
317 min = la::min(p1, p2);
318 max = la::max(p1, p2);
319 }
320
324 constexpr vec3 Size() const { return max - min; }
325
329 constexpr vec3 Center() const { return 0.5 * (max + min); }
330
335 constexpr double Scale() const {
336 vec3 absMax = la::max(la::abs(min), la::abs(max));
337 return la::max(absMax.x, la::max(absMax.y, absMax.z));
338 }
339
343 constexpr bool Contains(const vec3& p) const {
344 return la::all(la::gequal(p, min)) && la::all(la::gequal(max, p));
345 }
346

351 return la::all(la::gequal(box.min, min)) &&
352 la::all(la::gequal(max, box.max));
353 }
354
358 constexpr bool operator==(const Box& box) const {
359 return la::all(la::equal(box.min, min)) && la::all(la::equal(max, box.max));
360 }
361
365 constexpr bool operator!=(const Box& box) const { return !(*this == box); }

370 void Union(const vec3 p) {
371 min = la::min(min, p);
372 max = la::max(max, p);
373 }

381 out.max = la::max(max, box.max);

392 constexpr Box Transform(const mat3x4& transform) const {
393 Box out;

442 constexpr bool DoesOverlap(const Box& box) const {
443 return min.x <= box.max.x && min.y <= box.max.y && min.z <= box.max.z &&
444 max.x >= box.min.x && max.y >= box.min.y && max.z >= box.min.z;
445 }
