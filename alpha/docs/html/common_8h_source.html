109inline double sind(double x) {
110 if (!la::isfinite(x)) return NAN;
111 if (x < 0.0) return -sind(-x);
112#if defined(_WIN32) && defined(_M_ARM64)
113 x = std::fmod(x, 360.0);
114 const int quo = static_cast<int>(std::floor(x / 90.0 + 0.5));
115 x -= quo * 90.0;
116#else
117 int quo;
118 x = std::remquo(std::fabs(x), 90.0, &quo);
119#endif
120 const double xr = radians(x);
121 switch (quo % 4) {
122 case 0:
123 return math::sin(xr);
124 case 1:
125 return math::cos(xr);
126 case 2:
127 return -math::sin(xr);
128 case 3:
129 return -math::cos(xr);
130 }
131 return 0.0;
132}
