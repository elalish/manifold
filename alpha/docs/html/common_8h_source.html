126inline double sind(double x) {
127 if (!la::isfinite(x)) return NAN;
128 if (x < 0.0) return -sind(-x);
129#if defined(_WIN32) && defined(_M_ARM64)
130 x = std::fmod(x, 360.0);
131 const int quo = static_cast<int>(std::floor(x / 90.0 + 0.5));
132 x -= quo * 90.0;
133#else
134 int quo;
135 x = std::remquo(std::fabs(x), 90.0, &quo);
136#endif
137 const double xr = radians(x);
138 switch (quo % 4) {
139 case 0:
140 return math::sin(xr);
141 case 1:
142 return math::cos(xr);
143 case 2:
144 return -math::sin(xr);
145 case 3:
146 return -math::cos(xr);
147 }
148 return 0.0;
149}
150
156inline double cosd(double x) {
157 if (!la::isfinite(x)) return NAN;
158 return sind(90.0 - std::fmod(std::fabs(x), 360.0));
159}
