348 std::make_shared<std::mutex>();
349 mutable std::shared_ptr<CsgNode> pNode_;
350 // Optional attached ExecutionContext. shared_ptr so the Impl outlives
351 // the user's ExecutionContext if a ctx-attached Manifold survives it.
352 // Propagates through copy ctor / op= (raw copy preserves the attachment).
353 // Manifold-returning ops do *not* propagate it: derived Manifolds get a
354 // null ctx_. Eager ops (Status, WithPrecision, GetPrecision, Refine family,
355 // Hull, Minkowski) snapshot ctx_ to observe their in-call work; the snapshot
356 // uses an atomic load, which pins the Impl across long-running evaluations
357 // even if a concurrent op= reseats ctx_ mid-eval.
358 //
359 // Accessed only atomically (AtomicLoadShared/AtomicStoreShared): no const
360 // method mutates ctx_, but op= and the copy ctor write it on a Manifold that
361 // may be concurrently observed by const methods on other threads. The
362 // atomic-shared-ptr free functions give a torn-read-free snapshot
363 // without taking a lock. (pNode_ uses a mutex instead because lazy CSG
364 // eval mutates it through const methods, which an atomic load can't model.)
365 std::shared_ptr<ExecutionContext::Impl> ctx_;
366
367 std::shared_ptr<CsgNode> LoadPNode() const;
368 CsgLeafNode& GetCsgLeafNode(ExecutionContext::Impl* ctx = nullptr) const;
